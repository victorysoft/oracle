--物理路径 
select tablespace_name,
       file_name,
       file_id,
       round(bytes/(1024*1024),2) 空间M,
       online_status
       from dba_data_files;
       
--查询表空间使用情况
select a.tablespace_name 表空间名,
       a.file_name 物理路径,
       全部空间M,
       剩余空间M,
       已用空间M,
       (round(已用空间M/全部空间M,2)*100)||'%' 使用率 
       from (select a.tablespace_name ,
                    a.file_name ,
                    sum(round(a.bytes/(1024*1024),2)) 全部空间M,
                    sum(round(b.bytes/(1024*1024),2)) 剩余空间M,
                    sum(round(a.bytes/(1024*1024),2)-round(b.bytes/(1024*1024),2)) 已用空间M
                    from dba_data_files a
                    left join dba_free_space b
                      on b.tablespace_name = a.tablespace_name
                    group by a.tablespace_name,a.file_name
             ) a;
--创建表空间
